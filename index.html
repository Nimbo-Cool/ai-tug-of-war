async function callGeminiAI(promptText, retries = 3, delay = 2000) { saveApiKeyFromModal(); if (!apiKey) { throw new Error("請先在上方框框中輸入或貼上 Gemini API Key！"); } const url = `https://generativelanguage.googleapis.com/v1beta/models/gemini-3.8-flash:generateContent?key=${apiKey}`; for (let i = 0; i < retries; i++) { try { const response = await fetch(url, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ contents: [{ parts: [{ text: promptText }] }], generationConfig: { responseMimeType: "application/json" } }) }); if (response.status === 401) { throw new Error("API Key 無效或已過期，請重新貼上有效的 Gemini API Key。"); } if (!response.ok) { const errorData = await response.json().catch(() => ({})); const errMsg = errorData.error?.message || `HTTP ${response.status}`; // 若遇伺服器忙碌或流量過高，自動等待並重試 if (i < retries - 1) { await new Promise(res => setTimeout(res, delay)); continue; } throw new Error(`API 請求錯誤 (${errMsg})`); } const data = await response.json(); const rawText = data.candidates[0].content.parts[0].text; return JSON.parse(rawText); } catch (err) { if (i === retries - 1) throw err; await new Promise(res => setTimeout(res, delay)); } } }
